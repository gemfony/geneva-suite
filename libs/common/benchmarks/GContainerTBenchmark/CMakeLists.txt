SET ( EXECUTABLENAME GContainerTBenchmark )

SET ( ${EXECUTABLENAME}_SRCS
	GContainerTBenchmark.cpp
)

ADD_EXECUTABLE(${EXECUTABLENAME}
	${${EXECUTABLENAME}_SRCS}
)

SET(LIBRARIES_TO_LINK
		${GENEVA_LIBRARIES}
		${Boost_LIBRARIES})

# Add MPI Libraries in case building with MPI-consumer
IF(GENEVA_BUILD_WITH_MPI_CONSUMER)
	SET(LIBRARIES_TO_LINK
			${LIBRARIES_TO_LINK}
			${MPI_LIBRARIES})
ENDIF()

TARGET_LINK_LIBRARIES (${EXECUTABLENAME}
		${LIBRARIES_TO_LINK})

# NOTE: This benchmark is deliberately NOT registered via ADD_TEST() so that
# ctest does not run it during normal test runs. Run it manually instead:
#   ./GContainerTBenchmark --mode all --container all --quick

INSTALL ( TARGETS ${EXECUTABLENAME}
	DESTINATION ${INSTALL_PREFIX_DATA}/benchmarks/common/GContainerTBenchmark )
